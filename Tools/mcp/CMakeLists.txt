cmake_minimum_required(VERSION 3.22)
project(OrganicMcpTools VERSION 1.0.0 LANGUAGES C CXX)
set(JUCE_MODULES_ONLY ON CACHE BOOL "Native tools do not require JUCE packaging helpers" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(JUCE_PATH "D:/Projects/Dev/JUCE" CACHE PATH "Existing shared JUCE checkout")
if(NOT EXISTS "${JUCE_PATH}/CMakeLists.txt")
    message(FATAL_ERROR "Set JUCE_PATH to an existing shared JUCE checkout")
endif()
add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/shared-juce" EXCLUDE_FROM_ALL)
set(ORGANIC_MCP "${CMAKE_CURRENT_SOURCE_DIR}/../../Modules/juce_organicui/mcp")
add_library(organic_mcp_core STATIC "${ORGANIC_MCP}/McpCore.cpp" "${ORGANIC_MCP}/McpServer.cpp" "${ORGANIC_MCP}/../assistant/AssistantCore.cpp")
target_compile_features(organic_mcp_core PUBLIC cxx_std_17)
target_include_directories(organic_mcp_core PUBLIC "${ORGANIC_MCP}" "${CMAKE_CURRENT_SOURCE_DIR}/../../Modules" "${CMAKE_CURRENT_SOURCE_DIR}/../../External/asio")
target_compile_definitions(organic_mcp_core PUBLIC JUCE_WEB_BROWSER=0 JUCE_MODAL_LOOPS_PERMITTED=1)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # JUCE's raw Linux HTTP fallback pads POST bodies and blocks on keep-alive.
    # libcurl is already a supported Chataigne runtime dependency, including RPi.
    target_compile_definitions(organic_mcp_core PUBLIC JUCE_USE_CURL=1 JUCE_LOAD_CURL_SYMBOLS_LAZILY=1)
else()
    target_compile_definitions(organic_mcp_core PUBLIC JUCE_USE_CURL=0)
endif()
target_link_libraries(organic_mcp_core PUBLIC juce::juce_core juce::juce_events juce::juce_cryptography)
enable_testing()
add_executable(mcp_core_test ../tests/mcp_core_test.cpp)
target_link_libraries(mcp_core_test PRIVATE organic_mcp_core)
add_test(NAME mcp_core COMMAND mcp_core_test)
add_executable(mcp_lifecycle_test ../tests/mcp_lifecycle_test.cpp)
target_link_libraries(mcp_lifecycle_test PRIVATE organic_mcp_core)
add_test(NAME mcp_lifecycle COMMAND mcp_lifecycle_test)
set_tests_properties(mcp_lifecycle PROPERTIES TIMEOUT 30)
add_executable(assistant_core_test ../tests/assistant_core_test.cpp)
target_link_libraries(assistant_core_test PRIVATE organic_mcp_core)
add_test(NAME assistant_core COMMAND assistant_core_test)
add_library(organic_assistant_provider STATIC "${ORGANIC_MCP}/../assistant/AssistantProvider.cpp" "${ORGANIC_MCP}/../assistant/ClaudeProvider.cpp")
target_link_libraries(organic_assistant_provider PUBLIC organic_mcp_core)
add_executable(assistant_turn_test ../tests/assistant_turn_test.cpp)
target_link_libraries(assistant_turn_test PRIVATE organic_assistant_provider)
add_test(NAME assistant_turn COMMAND assistant_turn_test)
set_tests_properties(assistant_turn PROPERTIES TIMEOUT 45)
add_executable(assistant_subscription_smoke assistant_smoke.cpp)
target_link_libraries(assistant_subscription_smoke PRIVATE organic_assistant_provider)
add_executable(organic_mcp_example example.cpp)
target_link_libraries(organic_mcp_example PRIVATE organic_mcp_core)
add_executable(chataigne-mcp-bridge bridge.cpp)
target_link_libraries(chataigne-mcp-bridge PRIVATE organic_mcp_core)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME mcp_http COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tests/mcp_http_test.py" "$<TARGET_FILE:organic_mcp_example>")
    set_tests_properties(mcp_http PROPERTIES TIMEOUT 40)
    add_test(NAME mcp_bridge COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tests/mcp_bridge_test.py" "$<TARGET_FILE:organic_mcp_example>" "$<TARGET_FILE:chataigne-mcp-bridge>")
    set_tests_properties(mcp_bridge PROPERTIES TIMEOUT 50)
endif()
