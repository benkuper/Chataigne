cmake_minimum_required(VERSION 3.16)
project(controller_regressions LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(joycon_session_test joycon_session_test.cpp "${REPO}/Source/Module/modules/controller/joycon/JoyConDeviceSession.cpp")
target_include_directories(joycon_session_test PRIVATE "${REPO}/External/joycon/include" "${REPO}/Source/Module/modules/controller/joycon")
target_link_libraries(joycon_session_test PRIVATE Threads::Threads)
add_test(NAME joycon_session COMMAND joycon_session_test)
add_executable(joyshock_transport_test joyshock_transport_test.cpp "${REPO}/External/joycon/src/JoyShockLibrary.cpp")
target_include_directories(joyshock_transport_test PRIVATE "${REPO}/External/joycon/include" "${REPO}/External/joycon/src")
target_link_libraries(joyshock_transport_test PRIVATE Threads::Threads)
add_test(NAME joyshock_transport COMMAND joyshock_transport_test)
if(NOT MSVC)
  foreach(test joycon_session_test joyshock_transport_test)
    target_compile_options(${test} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${test} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()

if(APPLE)
  enable_language(OBJC)
  file(GLOB WIIUSE_SOURCES "${REPO}/Source/Module/modules/controller/wiimote/lib/wiiuse/*.c")
  add_executable(wiimote_mac_test wiimote_mac_test.m ${WIIUSE_SOURCES}
    "${REPO}/Source/Module/modules/controller/wiimote/lib/wiiuse/os_mac/os_mac.c"
    "${REPO}/Source/Module/modules/controller/wiimote/lib/wiiuse/os_mac/os_mac.m"
    "${REPO}/Source/Module/modules/controller/wiimote/lib/wiiuse/os_mac/os_mac_interface.m")
  target_include_directories(wiimote_mac_test PRIVATE "${REPO}/External/wiiuse")
  target_link_libraries(wiimote_mac_test PRIVATE "-framework Foundation" "-framework IOBluetooth")
  target_compile_options(wiimote_mac_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(wiimote_mac_test PRIVATE -fsanitize=address,undefined)
  add_test(NAME wiimote_mac COMMAND wiimote_mac_test)
endif()
